```{r}
# Cargar la tabla de superstores sales
superstore <- readr::read_csv(
  "../../data/superstore_sales.csv",

  show_col_types = FALSE
)

names(superstore)
```

```{r}
dplyr::glimpse(superstore)
```

```{r}
con <- DBI::dbConnect( 
    RSQLite::SQLite(), 
    ":memory:" 
) 

DBI::dbWriteTable( 
    con, 
    "superstore", 
    superstore, 
    overwrite = TRUE 
)

DBI::dbListTables(con) # produce:
```


