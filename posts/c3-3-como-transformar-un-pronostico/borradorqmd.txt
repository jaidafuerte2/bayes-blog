```{r}
superstore %>% summarise( 
    filas = n(), 
    fechas_na = sum(is.na(order_date)), 
    fecha_minima = min(order_date, na.rm = TRUE),

)
```

```{r}
# Agrupar las ventas por mes
ventas_mensuales <- superstore %>% 
  mutate( 
      mes = floor_date(order_date, unit = "month") 
  ) %>% 
  group_by(mes) %>% 
  summarise( 
      ventas = sum(Sales, na.rm = TRUE), 
      .groups = "drop" 
  ) %>% 
  arrange(mes) 

ventas_mensuales
```

```{r}
# Visualizar la evolución histórica de las ventas
```

```{r}
# Resumir las 1000 predicciones posteriores en los percentiles
# 10, 50 y 90
escenarios <- tibble( 
    periodo = seq_len(horizonte), 
    demanda_baja = apply( 
        predicciones, 
        2, 
        quantile, 
        probs = 0.10 
    ), 
    demanda_central = apply( 
        predicciones, 
        2, 
        quantile, 
        probs = 0.50 
    ), 
    demanda_alta = apply( 
        predicciones, 
        2, 
        
        quantile, probs = 0.90 
    ) 
)
escenarios
```

```{r}
# Transformar los escenarios de los 3 meses posteriores a su 
# formato largo
escenarios_largo <- escenarios %>% 
  pivot_longer( 
    cols = c( 
        demanda_baja, 
        demanda_central, 
        demanda_alta 
    ), 
    names_to = "escenario", 
    values_to = "ventas" 
)
escenarios_largo
```

```{r}
# Graficar tres escenarios para la demanda futura de
# los próximos 3 meses
ggplot( 
    escenarios_largo,
    aes(x = periodo, y = ventas, linetype = escenario) 
) + 
  geom_line(linewidth = 1) + 
  geom_point(size = 2) + 
  scale_x_continuous( 
      breaks = seq_len(horizonte) 
  ) + 
  labs( 
      title = "Tres escenarios para la demanda futura", 
      x = "Periodo futuro", 
      y = "Ventas previstas", 
      linetype = "Escenario" 
  ) + 
  theme_minimal()
```
