```{r}
ventas_mensuales <- dbGetQuery( 
    con, 
    " 
    SELECT 
      strftime(
      '%Y-%m-01',
      date('1970-01-01', order_date || ' days')
      ) AS period,
      SUM(Sales) AS sales, 
      AVG(Sales) AS average_sale, 
      COUNT(*) AS transactions 
    FROM superstore 
    WHERE order_date IS NOT NULL 
    GROUP BY period 
    ORDER BY period 
    " 
)
ventas_mensuales[1:5,]
```

```r
modelo <- stan_glm(
```

```{r}
pred_resumen <- tibble( 
    period = test$period, 
    observed = test$sales, 
    predicted_median = apply(predicciones, 2, median), 
    lower_90 = apply(predicciones, 2, quantile, probs = 0.05), 
    upper_90 = apply(predicciones, 2, quantile, probs = 0.95) 
)
pred_resumen
```

```{r}
ggplot(pred_resumen, aes(x = period)) + 
  geom_ribbon( 
      aes(ymin = lower_90, 
          ymax = upper_90), 
      alpha = 0.20 ) +

      aes(y = observed), 
      linewidth = 1
```

```{r}
metricas <- pred_resumen %>% 
  summarise(
      MAE = mean(abs(observed - predicted_median)), 
      RMSE = sqrt(mean((observed - predicted_median)^2)), 
      cobertura_90 = mean( 
          observed >= lower_90 & 
          observed <= upper_90 
      ) 
  )
metricas
```
