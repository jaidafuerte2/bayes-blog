```{r}
# Crear serie mensual 
ventas_mensuales <- superstore |> 
  filter(!is.na(Order_Date)) |> 
  mutate( 
      Mes = floor_date( 
          Order_Date, unit = "month" 
      ) 
  ) |> 
  group_by(Mes) |> 
  summarise( 
      Sales = sum( 
          Sales, na.rm = TRUE 
      ), 
      .groups = "drop" 
  ) |> 
  arrange(Mes)
ventas_mensuales
```

```{r}
# Crear índice temporal 
ventas_mensuales <- ventas_mensuales |> 
  mutate( 
      Tiempo = row_number() 
  )
ventas_mensuales
```
